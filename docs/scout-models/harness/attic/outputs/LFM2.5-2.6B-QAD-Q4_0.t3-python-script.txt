#!/usr/bin/env python3
import sys
import csv

def main():
    if len(sys.argv) != 2:
        sys.stderr.write("Usage: script.py <csv_file>\n")
        sys.exit(1)

    filepath = sys.argv[1]

    try:
        with open(filepath, newline='') as f:
            reader = csv.DictReader(f)
            if 'status' not in reader.fieldnames:
                sys.stderr.write("Error: missing 'status' column\n")
                sys.exit(1)
            counts = {}
            for row in reader:
                status = row['status']
                counts[status] = counts.get(status, 0) + 1
            for status, cnt in sorted(counts.items(), key=lambda x: x[1], reverse=True):
                print(f"{status}: {cnt}")
    except Exception as e:

        sys.exit(1)
